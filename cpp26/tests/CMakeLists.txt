include(GoogleTest)
include(Clone26Targets)

# The designs. Design A is the hook mixin, B is std::polymorphic, C is the reflection mixin. The delete_reason
# target is its own, because = delete("reason") is a C++26 feature a compiler may lack on its own.
clone26_add_target(design_a_tests TEST
  SOURCES design_a_test.cpp
  REQUIRES DEDUCING_THIS)

clone26_add_target(delete_reason_tests TEST
  SOURCES delete_reason_test.cpp
  REQUIRES DELETE_REASON)

clone26_add_target(design_b_tests TEST
  SOURCES design_b_test.cpp
  REQUIRES POLYMORPHIC)

clone26_add_target(design_c_tests TEST
  SOURCES design_c_test.cpp
  REQUIRES DEDUCING_THIS REFLECTION)

# The guard for the requirement of design C: two translation units of one executable export the classes that
# clone() searches, as each of them sees them, and a third compares the lists. Its mutant twin compiles translation
# unit B against a copy of figure_reflected.hpp with one more class in shapes_reflected, so the lists differ and the
# comparison fails. The twin shows the violation only through the comparison: no file of the guard calls clone(), so
# the twin never runs the dispatch that the violation would break. Like the other twins it is left out of the
# default build and of CTest.
clone26_add_target(design_c_odr TEST
  SOURCES design_c_odr_test.cpp design_c_tu_a.cpp design_c_tu_b.cpp
  REQUIRES DEDUCING_THIS REFLECTION
  MUTANT
  MUTANT_INCLUDE_DIR ${CLONE26_MUTANT_EXTRA_CLASS_INCLUDE_DIR}
  MUTANT_SOURCES design_c_tu_b.cpp)

# The nets. Each one checks the shipped hierarchy of design A, in which ForgetfulTile forgot its overrides and Tile
# did not: the nets have to see ForgetfulTile, and a Tile that loses its overrides has to fail them. Each net has a
# mutant twin that compiles it against a header in which Tile lost its overrides, and check_nets.sh expects the
# twins to fail. net_constexpr and net_lint are built and never run. The others are run.
#
# Net 1 is the contract net, in two targets. Only they use the contracts flavour of clone_check.hpp, and each picks
# its evaluation semantic with a flag. net_contract replaces the violation handler with a counting one and
# observes. net_contract_death enforces, and a death test sees the process terminate.
clone26_add_target(net_contract TEST
  SOURCES net_contract_test.cpp
  REQUIRES DEDUCING_THIS CONTRACTS CONTRACT_OBSERVE_FLAG
  LINK clone26_contracts
  OPTIONS ${CLONE26_CONTRACT_OBSERVE_FLAG}
  MUTANT)

clone26_add_target(net_contract_death TEST
  SOURCES net_contract_death_test.cpp
  REQUIRES DEDUCING_THIS CONTRACTS CONTRACT_ENFORCE_FLAG
  LINK clone26_contracts
  OPTIONS ${CLONE26_CONTRACT_ENFORCE_FLAG}
  MUTANT)

# Net 2: a constant-evaluated round trip.
clone26_add_target(net_constexpr OBJECT
  SOURCES net_constexpr.cpp
  REQUIRES DEDUCING_THIS CONSTEXPR_ROUND_TRIP
  MUTANT)

# Net 3a: a reflection lint. Net 3b: tests generated by reflection.
clone26_add_target(net_lint OBJECT
  SOURCES net_lint.cpp
  REQUIRES DEDUCING_THIS REFLECTION
  MUTANT)

clone26_add_target(net_generated TEST
  SOURCES net_generated_test.cpp
  REQUIRES DEDUCING_THIS REFLECTION
  MUTANT)
